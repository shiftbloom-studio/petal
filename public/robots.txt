# Only the start page is meant to be found. Chat pages show the messages of a channel, and
# crawlers that ignore this file are refused there. The rules are the same for search engines
# and for the crawlers of language models, which are welcome on the start page too.
# Content-Signal (https://contentsignals.org): search and AI answers yes, AI training no.
User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /$
# Petal in plain text, for language models (https://llmstxt.org).
Allow: /llms.txt
Allow: /llms-full.txt
Allow: /sitemap.xml
# What the start page is made of. Without its styles, scripts and fonts a search engine
# renders a broken page, and without the image a shared link has no preview.
Allow: /_build/
Allow: /fonts/
Allow: /og.png
Allow: /og-square.png
Allow: /bloom.svg
Allow: /favicon.ico
Allow: /apple-touch-icon.png
Disallow: /
